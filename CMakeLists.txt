cmake_minimum_required(VERSION 3.20)
project(AstraStudio VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ASTRA_SANITIZERS "Build with ASan/UBSan (GCC/Clang)" OFF)
add_library(astra_engine STATIC src/engine.cpp)
target_include_directories(astra_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(MSVC)
    target_compile_options(astra_engine PRIVATE /W4 /permissive-)
else()
    target_compile_options(astra_engine PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(ASTRA_SANITIZERS AND NOT MSVC)
    target_compile_options(astra_engine PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(astra_engine PUBLIC -fsanitize=address,undefined)
endif()
add_executable(astra-studio src/server.cpp)
target_link_libraries(astra-studio PRIVATE astra_engine)
if(WIN32)
    target_link_libraries(astra-studio PRIVATE ws2_32 shell32)
endif()
add_custom_command(TARGET astra-studio POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/web" "$<TARGET_FILE_DIR:astra-studio>/web")
enable_testing()
add_executable(astra_tests tests/test_engine.cpp)
target_link_libraries(astra_tests PRIVATE astra_engine)
add_test(NAME engine COMMAND astra_tests)
